# Called from code once a year for all characters, at a random point that year (individually for each character)
# Root is the character
random_yearly_everyone_pulse = {
	if = {
		limit = {
			health >= 1
			has_character_flag = had_low_health_permenant
		}
		remove_character_flag = had_low_health_permenant
		remove_character_flag = had_low_health_1yr
		remove_character_flag = had_low_health_2yr
		remove_character_flag = had_low_health_3yr
	}
	if = {
		limit = {
			health <= 0
		}
		add_character_flag = {
			flag = had_low_health_permenant
			years = 10
		}
		add_character_flag = {
			flag = had_low_health_1yr
			days = 400
		}
		add_character_flag = {
			flag = had_low_health_2yr
			days = 800
		}
		add_character_flag = {
			flag = had_low_health_3yr
			days = 1200
		}
	}
	if = {
		limit = {
			has_character_flag = had_low_health_permenant
			NOT = { has_character_flag = had_low_health_3yr }
		}
		death = natural
	}
	if = {
		limit = {
			has_character_flag = had_low_health_permenant
			has_character_flag = had_low_health_3yr
			NOT = { has_character_flag = had_low_health_2yr }
		}
		random = {
			chance = 75
			death = natural
		}
	}
	if = {
		limit = {
			has_character_flag = had_low_health_permenant
			has_character_flag = had_low_health_3yr
			has_character_flag = had_low_health_2yr
			NOT = { has_character_flag = had_low_health_1yr }
		}
		random = {
			chance = 50
			death = natural
		}
	}
	if = {
		limit = {
			has_character_flag = had_low_health_permenant
			has_character_flag = had_low_health_3yr
			has_character_flag = had_low_health_2yr
			has_character_flag = had_low_health_1yr
		}
		random = {
			chance = 25
			death = natural
		}
	}
}
