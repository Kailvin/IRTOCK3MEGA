# UBA奇观flag
uba_remove_uba_unique_flags_building_effect = {
	if = {
		limit = {
			is_target_in_global_variable_list = {
				name = uba_unique_buildings_finished
				target = flag:$BUILDING$_finished
			}
		}
		remove_list_global_variable = {
			name = uba_unique_buildings_finished
			target = flag:$BUILDING$_finished
		}
	}
}
# UBA矿井flag
uba_remove_uba_unique_flags_mine_effect = {
	if = {
		limit = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:$MINE$_mines_added
			}
        }
		remove_list_global_variable = {
			name = unavailable_unique_decisions
			target = flag:$MINE$_mines_added
		}
	}
}
