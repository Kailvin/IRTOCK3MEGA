	switch = {
		trigger = has_building
		$BUILDING$_01 = { remove_building = $BUILDING$_01 }
		$BUILDING$_02 = { remove_building = $BUILDING$_02 }
		$BUILDING$_03 = { remove_building = $BUILDING$_03 }
		$BUILDING$_04 = { remove_building = $BUILDING$_04 }
		$BUILDING$_05 = {
			remove_building = $BUILDING$_05
			if = {
				limit = {
					is_target_in_global_variable_list = {
						name = uba_unique_buildings_finished
						target = flag:$BUILDING$_finished
					}
				}
				remove_list_global_variable = {
					name = uba_unique_buildings_finished
					target = flag:$BUILDING$_finished
				}
			}
		}
	}
}
