	NOT = { has_game_rule = uba_common_buildings_off }
	OR = {
		scope:holder = {
			any_directly_owned_province = {
				OR = {
					trigger_if = { 
						limit = { has_global_variable = is_uba_loaded }
						OR = {
							has_building_or_higher = UBA_reserve_03
							has_building_or_higher = UBA_test_building_1
						}
					}
				}
			}
		}
		trigger_if = { 
			limit = { has_global_variable = is_rba_loaded }
			culture = { has_innovation = innovation_rba_private_estate }
			OR = {
				terrain = farmlands
				terrain = plains
				terrain = drylands
				terrain = floodplains
				terrain = oasis
				county = {
					OR = {
						has_county_modifier = fertile_desert_mountains_modifier
						has_county_modifier = dodekapolis_modifier
					}
				}
			}
		}
	}
}

# 葡萄园 种植园革新
building_uba_vineyard_requirement = {
	NOT = { has_game_rule = uba_common_buildings_off }
	OR = {
		scope:holder = {
			any_directly_owned_province = {
				OR = {
					trigger_if = { 
						limit = { has_global_variable = is_uba_loaded }
						OR = {
							has_building_or_higher = UBA_test_building_1
							has_building_or_higher = UBA_garden_03
						}
					}
				}
			}
		}
		AND = {
			has_global_variable = is_rba_loaded
			NOR = { # 地形
				terrain = desert
				terrain = desert_mountains
				terrain = taiga
				terrain = jungle
				terrain = steppe
			}
			OR = { # 非葡萄产地
				has_game_rule = rba_region_limit_off
				NOT = { geographical_region = rba_no_vineyard_region }
			}
			OR = { # 文化与宗教
				AND = {
					has_building_or_higher = temple_$NUMBER$
					OR = {
						faith = faith:catholic
						faith = { has_doctrine_parameter = allows_brewery }
					}
				}
				AND = {
					has_building_or_higher = city_$NUMBER$
					culture = { has_cultural_parameter = allows_brewery }
				}
				AND = {
					trigger_if = { 
						limit = { has_global_variable = is_rba_loaded }
						culture = { has_cultural_parameter = unlocks_breweries_for_other_holdings }
					}
					OR = {
						has_building_or_higher = city_$NUMBER$
						has_building_or_higher = temple_$NUMBER$
						has_building_or_higher = castle_$NUMBER$
						has_building_or_higher = temple_citadel_$NUMBER$
					}
					# scope:holder = {
					# 	OR = {
					# 		NOT = { religion = religion:islam_religion }
					# 		has_trait = drunkard
					# 		has_trait = lifestyle_reveler
					# 	}
					# }
				}
			}
		}
	}
}

			}
		}
	}

				}
			}
			scope:holder = {
				any_directly_owned_province = {
					OR = {
						has_building_or_higher = UBA_garden_03
						has_building_or_higher = UBA_test_building_1
					}
				}
			}
		}
	}
	trigger_else_if = {
		limit = { has_global_variable = is_uba_loaded }
		scope:holder = {
			any_directly_owned_province = {
				OR = {
					has_building_or_higher = UBA_garden_03

				}
			}
		}
	}
