uba_nomadic_building_barony_is_valid_trigger = {
	holder = $CHARACTER$

	title_province = {
		has_special_building = yes
		OR = {
			AND = {
				has_building_or_higher = UBA_hunting_01
				NOT = { has_building = UBA_hunting_05 }
				NAND = {
					has_building = UBA_hunting_04
					is_target_in_global_variable_list = {
						name = uba_unique_buildings_finished
						target = flag:UBA_hunting_finished
					}
				}
			}
			AND = {
				has_building_or_higher = UBA_pasture_01
				NOT = { has_building = UBA_pasture_05 }
				NAND = {
					has_building = UBA_pasture_04
					is_target_in_global_variable_list = {
						name = uba_unique_buildings_finished
						target = flag:UBA_pasture_finished
					}
				}
				
			}
			AND = {
				has_building_or_higher = UBA_yurt_01
				NOT = { has_building = UBA_yurt_05 }
				NAND = {
					has_building = UBA_yurt_04
					is_target_in_global_variable_list = {
						name = uba_unique_buildings_finished
						target = flag:UBA_yurt_finished
					}
				}
			}
		}
	}
	NOT = { is_leased_out = yes }
}
