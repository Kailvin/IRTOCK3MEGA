namespace = liao_invasion

### Spawn invader with the associated story...
liao_invasion.0001 = {
	scope = none
	hidden = yes
	
	trigger = {
		exists = title:e_liao # update compatibility - new titles in old saves return false
		any_county_in_region = {
			region = special_liao_conquest_region_start
			count > 0
			holder = { is_ai = yes }
		}
	}

	on_trigger_fail = {
		debug_log = "liao story has failed to trigger; trying again next year"
		debug_log_date = yes
	}

	immediate = {
		debug_log = "liao story owner has appeared"
		debug_log_date = yes
		
		character:nn_yelu_nulisi = { save_scope_as = nolisi_liao }

		scope:nolisi_liao.dynasty = {
			# pick an already existing dynasty member
			ordered_dynasty_member = {
				limit = {
					is_landed = yes
					highest_held_title_tier > tier_barony
					OR = {
						capital_county = { title_province = { geographical_region = special_liao_conquest_region_total } }
						capital_county = { title_province = { geographical_region = world_asia_north_east } }
					}
					is_independent_ruler = yes
					is_adult = yes
					is_healthy = yes
					is_imprisoned = no
				}
				alternative_limit = {
					is_landed = yes
					highest_held_title_tier > tier_barony
					OR = {
						capital_county = { title_province = { geographical_region = special_liao_conquest_region_total } }
						capital_county = { title_province = { geographical_region = world_asia_north_east } }
					}
					is_adult = yes
					is_healthy = yes
					is_imprisoned = no
				}
				order_by = realm_size
				save_scope_as = invader
			}
		}
		
		# only spawns a new character if scope:invader is not set
		spawn_liao_invader_effect = {
			INVADER_DYNN = liao
		}

		# start story
		scope:invader = {
			create_story = story_liao_invasion
		}
	}
}

# Event for handling army spawning
liao_invasion.0100 = {
	hidden = yes
	immediate = {
		spawn_liao_troops_effect = yes
	}
}

# Event for declaring the very first war; values like current_ / max_military_strength aren't immediately filled with a value and return 0 instead
liao_invasion.0200 = {
	hidden = yes
	trigger = { is_ai = yes } # don't force a player into a war
	immediate = {
		war_target_evaluation_and_declaration_effect = {
			INVADER_DYNN = liao
			INVADER_CHAR = scope:story.story_owner
			INVADER_REGION = total
			INVADER_TARGET_TITLE_TIER = kingdom
			INVADER_RANGE = neighboring
			INVADER_LIEGE_RESTRICTIONS = ""
		}
	}
}

### The invader has appeared! Would you like to play as them?
liao_invasion.1001 = { # only players see this
	type = character_event
	title = liao_invasion.1001.t
	desc = liao_invasion.1001.desc
	
	theme = hi_theme_steppe

	left_portrait = {
		character = scope:invader
		animation = personality_bold
	}

	trigger = { hi_show_invader_notification_event_trigger = yes }
	
	option = { name = liao_invasion.1001.a }

	option = { # play as the newly spawned invader
		name = liao_invasion.1001.b
		trigger = { hi_play_as_invader_event_option_trigger = yes }
		add_internal_flag = dangerous
		scope:invader = {
			set_variable = {
				name = historicinvasions_previous_ruler_var
				value = root
				months = 1
			}
		}
		set_player_character = scope:invader
	}
}

### invader gets the option to convert their faith
liao_invasion.3001 = {
	type = character_event
	title = liao_invasion.3001.t
	desc = liao_invasion.3001.desc
	
	theme = hi_theme_steppe

	left_portrait = {
		character = root
		animation = personality_bold
	}

	trigger = {
		faith = { has_doctrine_parameter = unreformed }
	}
	
	option = { # convert
		name = liao_invasion.3001.a
		add_internal_flag = dangerous
		set_character_faith_with_conversion = faith:mahayana
		ai_chance = { base = 100 }
	}

	option = { # don't convert
		name = liao_invasion.3001.b
		add_piety = 50
		ai_chance = { base = 0 }
	}
}
